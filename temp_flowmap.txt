const FlowMapScreen = ({ go, showHints, variant, onVariantChange }) => {
  const [progress, setProgress] = useWalkProgress(true);
  const eta = "10:58";
  useEffect3(() => { if (progress >= 1) setTimeout(() => go("indoor"), 600); }, [progress]);
  const variants = ["Floating", "Steps", "Big ETA"];
  return (
    <div className="phone">
      <div className="phone-inner">
        <div className="statusbar"><span>10:54</span><span>••• 􀟝􀛨</span></div>
        <div className="bb-chrome">
          <span className="bb-tag">≡ Blackboard</span>
          <span style={{ flex: 1 }}>→ STAT2003 · 50-N202</span>
          <Hotspot label="map:cancel" onTap={() => go("timetable")}>
            <span style={{ pointerEvents: "none" }}>×</span>
          </Hotspot>
        </div>
        {variant === "Floating" && <MapFloating progress={progress} eta={eta} />}
        {variant === "Steps" && <MapStepList progress={progress} eta={eta} />}
        {variant === "Big ETA" && <MapBigEta progress={progress} eta={eta} />}

        <div style={{ padding: "8px 12px", display: "flex", gap: 8 }}>
          <Hotspot label={`map:${variant}:skip-to-end`} onTap={() => go("indoor")} style={{ flex: 1 }}>
            <button className="btn ghost" style={{ width: "100%", pointerEvents: "none" }}>Skip to building →</button>
          </Hotspot>
        </div>
        <div className="variant-strip">
          {variants.map(v => (
            <button key={v} className={"variant-pill " + (variant === v ? "active" : "")}
              onClick={() => { onVariantChange(v); setProgress(0); }}>{v}</button>
          ))}
        </div>
      </div>
    </div>
  );
};

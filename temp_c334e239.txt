function useWalkProgress(running) {
  const [progress, setProgress] = useState3(0);
  useEffect3(() => {
    if (!running) return;
    const id = setInterval(() => setProgress(p => Math.min(1, p + 0.012)), 120);
    return () => clearInterval(id);
  }, [running]);
  return [progress, setProgress];
}

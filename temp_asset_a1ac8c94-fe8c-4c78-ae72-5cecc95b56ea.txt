// ---------- shared route svg ----------
function MiniRouteMap({ progress = 0, height = 180, showLabels = true }) {
  // path approx: 40,160 → 100,120 → 180,80 → 260,40
  const points = [[40, 160], [100, 120], [180, 80], [260, 40]];
  const t = Math.max(0, Math.min(1, progress));
  // interpolate along polyline
  const segs = [];
  let totalLen = 0;
  for (let i = 0; i < points.length - 1; i++) {
    const dx = points[i+1][0] - points[i][0], dy = points[i+1][1] - points[i][1];
    const len = Math.hypot(dx, dy);
    segs.push({ from: points[i], to: points[i+1], len });
    totalLen += len;
  }
  let target = t * totalLen;
  let pos = points[0];
  for (const s of segs) {
    if (target <= s.len) {
      const f = target / s.len;
      pos = [s.from[0] + (s.to[0] - s.from[0]) * f, s.from[1] + (s.to[1] - s.from[1]) * f];
      break;
    }
    target -= s.len;
    pos = s.to;
  }
  return (
    <svg viewBox="0 0 300 180" style={{ width: "100%", height, background: "var(--paper)", border: "1.5px dashed var(--pencil)", borderRadius: 8 }}>
      {/* topo */}
      {[0, 1, 2, 3, 4].map(i => (
        <path key={i} d={`M0 ${30 + i*32} Q150 ${20 + i*32} 300 ${30 + i*32}`} stroke="var(--pencil-light)" strokeWidth="0.8" fill="none" opacity="0.4" />
      ))}
      {/* buildings */}
      <rect x="20" y="140" width="40" height="30" fill="none" stroke="var(--pencil)" strokeWidth="1.2" />
      <rect x="240" y="20" width="40" height="30" fill="none" stroke="var(--pencil)" strokeWidth="1.2" />
      {/* path */}
      <polyline points="40,160 100,120 180,80 260,40" stroke="var(--pencil)" strokeWidth="2.5" strokeDasharray="5 4" fill="none" />
      <polyline points={`40,160 ${pos[0]},${pos[1]}`} stroke="var(--accent)" strokeWidth="3" fill="none" />
      {/* you-dot */}
      <circle cx={pos[0]} cy={pos[1]} r="8" fill="var(--accent)" stroke="white" strokeWidth="2" />
      <circle cx={pos[0]} cy={pos[1]} r="14" fill="none" stroke="var(--accent)" strokeWidth="1" opacity="0.4" />
      {/* labels */}
      {showLabels && (<>
        <text x="40" y="178" fontSize="9" fill="var(--pencil)">50-N202</text>
        <text x="220" y="14" fontSize="9" fill="var(--accent)" fontWeight="600">★ 78-217</text>
      </>)}
    </svg>
  );
}

// Slider value 0..1 maps to "minutes from now until departure" from 0 → 30
// "Now" is 10:42 (matches status bar). Class at 11:00 (start), 6 min walk.
// slider=0 → leave now (10:42) → arrive 10:48 → 12 min early ✓
// slider=1 → leave in 30 min (11:12) → arrive 11:18 → 18 min late ✗
function useEta(slider) {
  const minutesFromNow = Math.round(slider * 30); // 0 → 30
  const NOW_MIN = 42; // 10:42 from status bar
  const departMin = NOW_MIN + minutesFromNow; // minutes past 10:00
  const arriveMin = departMin + 6;
  const buffer = 60 - arriveMin; // positive = on time
  const onTime = buffer >= 0;
  const lateMin = Math.max(0, -buffer);
  const fmt = (m) => {
    const h = 10 + Math.floor(m / 60);
    const mm = Math.floor(m % 60);
    return `${h}:${String(mm).padStart(2, "0")}`;
  };
  return {
    departStr: fmt(departMin),
    arriveStr: fmt(arriveMin),
    onTime,
    lateMin,
    bufferMin: Math.max(0, buffer),
    minutesFromNow,
  };
}

    const r = trackRef.current.getBoundingClientRect();

const SliderD = ({ value, onChange, showHints }) => {
  // Dial: 30 min before class (top, slider=0) → 0 min before (right side, slider=1)
  // Map slider value (0..1) to angle on a 270° arc starting at -225° (top-left) ending at 45° (right)
  // Use a clean clock-like layout: 0 min at right (3 o'clock), 30 min at top (12 o'clock)
  // We'll go: slider=0 → -225° (top-left), slider=1 → 45° (bottom-right)... actually clean it up.
  // Simpler: arc from -135° (lower-left) clockwise through top to +135° (lower-right). 0 → 1.
  const eta = useEta(value);
  const startDeg = -225; // visual start (lower-left)
  const sweep = 270;     // total arc
  const angle = startDeg + value * sweep; // -225 → 45
  const cx = 110, cy = 110, r = 78;
  const rad = (deg) => (deg * Math.PI) / 180;
  const tipX = cx + r * Math.cos(rad(angle));
  const tipY = cy + r * Math.sin(rad(angle));
  const dialRef = useRef2(null);
  const [drag, setDrag] = useState2(false);

  const onPointer = (e) => {
    if (!dialRef.current) return;
    const rect = dialRef.current.getBoundingClientRect();
    const x = e.clientX - rect.left - rect.width / 2;
    const y = e.clientY - rect.top - rect.height / 2;
    let a = (Math.atan2(y, x) * 180) / Math.PI; // -180..180
    // normalize so a is in [-225, 45] arc range:
    // valid arc spans from -225° (=135°) clockwise to 45°
    // map atan2 result into that domain
    // atan2 range: -180..180. We want to treat <-180 wrap-around.
    // Convert any angle to [-225, 135] domain via shift:
    if (a > 45 && a <= 135) {
      // dead zone (top of arc gap, lower part missing) — clamp to nearest end
      // closer to 45 → snap to 45 (slider=1), closer to 135 (=-225) → snap to 0
      a = a < 90 ? 45 : -225;
    } else if (a > 135) {
      a = a - 360; // make it negative (e.g. 170 → -190)
    }
    // now a should be in [-225, 45]
    const t = (a - startDeg) / sweep;
    onChange(Math.max(0, Math.min(1, t)));
  };

  // tick marks every 5 minutes (so 0, 5, 10, 15, 20, 25, 30 min before class)
  // Layout: "now" at lower-left (slider=0), "30m" at lower-right (slider=1)
  const ticks = [0, 5, 10, 15, 20, 25, 30].map(m => {
    const v = m / 30; // m=0 → slider=0 → lower-left (now); m=30 → slider=1 → lower-right
    const a = startDeg + v * sweep;
    const major = m % 10 === 0;
    const inner = major ? r - 14 : r - 8;
    const outer = r - 2;
    return { m, a, major, inner, outer, label: m === 0 ? "now" : `${m}m` };
  });

  // active arc length
  const arcLen = (value * sweep * Math.PI * r) / 180;

  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 12, padding: "0 14px", flex: 1, alignItems: "center" }}>
      <MiniRouteMap progress={0} height={130} />
      <div
        ref={dialRef}
        onMouseDown={(e) => { setDrag(true); onPointer(e); }}
        onMouseMove={(e) => { if (drag) onPointer(e); }}
        onMouseUp={() => setDrag(false)}
        onMouseLeave={() => setDrag(false)}
        onTouchStart={(e) => { setDrag(true); const t = e.touches[0]; onPointer({ clientX: t.clientX, clientY: t.clientY }); }}
        onTouchMove={(e) => { if (drag) { const t = e.touches[0]; onPointer({ clientX: t.clientX, clientY: t.clientY }); } }}
        onTouchEnd={() => setDrag(false)}
        style={{ width: 220, height: 220, position: "relative", cursor: "pointer", touchAction: "none", userSelect: "none" }}
      >
        <svg viewBox="0 0 220 220" style={{ width: "100%", height: "100%" }}>
          {/* base track ring */}
          <path
            d={`M ${cx + r * Math.cos(rad(startDeg))} ${cy + r * Math.sin(rad(startDeg))}
                A ${r} ${r} 0 1 1 ${cx + r * Math.cos(rad(startDeg + sweep))} ${cy + r * Math.sin(rad(startDeg + sweep))}`}
            fill="none" stroke="var(--pencil-light)" strokeWidth="3" strokeLinecap="round" />
          {/* active arc */}
          <path
            d={`M ${cx + r * Math.cos(rad(startDeg))} ${cy + r * Math.sin(rad(startDeg))}
                A ${r} ${r} 0 1 1 ${cx + r * Math.cos(rad(startDeg + sweep))} ${cy + r * Math.sin(rad(startDeg + sweep))}`}
            fill="none"
            stroke={eta.onTime ? "var(--ok)" : "var(--warn)"}
            strokeWidth="6" strokeLinecap="round"
            strokeDasharray={`${arcLen} 1000`}
          />
          {/* tick marks + labels */}
          {ticks.map((t, i) => {
            const x1 = cx + t.inner * Math.cos(rad(t.a));
            const y1 = cy + t.inner * Math.sin(rad(t.a));
            const x2 = cx + t.outer * Math.cos(rad(t.a));
            const y2 = cy + t.outer * Math.sin(rad(t.a));
            const lx = cx + (r + 14) * Math.cos(rad(t.a));
            const ly = cy + (r + 14) * Math.sin(rad(t.a));
            return (
              <g key={i}>
                <line x1={x1} y1={y1} x2={x2} y2={y2}
                  stroke="var(--ink-soft)" strokeWidth={t.major ? 2 : 1} strokeLinecap="round" />
                {t.major && (
                  <text x={lx} y={ly + 3} textAnchor="middle" fontSize="9" fill="var(--pencil)">
                    {t.label}
                  </text>
                )}
              </g>
            );
          })}
          {/* needle */}
          <line x1={cx} y1={cy} x2={tipX} y2={tipY}
            stroke="var(--ink)" strokeWidth="3" strokeLinecap="round" />
          <circle cx={tipX} cy={tipY} r="11" fill="var(--accent)" stroke="white" strokeWidth="2.5"
            style={{ filter: showHints ? "drop-shadow(0 0 6px oklch(0.85 0.06 305))" : "none" }} />
          <circle cx={cx} cy={cy} r="3" fill="var(--ink)" />
          {/* center labels */}
          <text x={cx} y={cy - 14} textAnchor="middle" fontSize="9" fill="var(--pencil)">depart at</text>
          <text x={cx} y={cy + 14} textAnchor="middle" fontSize="22" fontWeight="700" fill="var(--ink)">{eta.departStr}</text>
          <text x={cx} y={cy + 30} textAnchor="middle" fontSize="10" fill="var(--pencil)">
            {eta.minutesFromNow === 0 ? "right now" : `in ${eta.minutesFromNow} min`}
          </text>
        </svg>
      </div>
      <div className="box" style={{ width: "100%", padding: 10, borderColor: eta.onTime ? "var(--ok)" : "var(--warn)", display: "flex", gap: 12, alignItems: "center" }}>
        <div style={{ flex: 1 }}>
          <div className="label">arrive at</div>
          <div style={{ fontSize: 26, fontWeight: 700, color: eta.onTime ? "var(--ok)" : "var(--warn)", lineHeight: 1 }}>{eta.arriveStr}</div>
          <div style={{ fontSize: 11, color: "var(--pencil)", marginTop: 4 }}>
            {eta.onTime ? `${eta.bufferMin} min buffer · on time` : `${eta.lateMin} min late`}
          </div>
        </div>
        <div className={"pill " + (eta.onTime ? "ok" : "warn")} style={{ fontSize: 13, padding: "6px 12px" }}>
          {eta.onTime ? "✓ on time" : "⚠ late"}
        </div>
      </div>
    </div>
  );
};
